document.addEventListener("DOMContentLoaded", function() {
    const restoButtons = document.querySelectorAll(".resto-menu-btn");

    restoButtons.forEach(btn => {
        btn.addEventListener("click", function() {
            const currentItem = this.parentElement;
            const statusLabel = this.querySelector(".resto-menu-status");
            
            document.querySelectorAll(".resto-menu-item").forEach(item => {
                if (item !== currentItem) {
                    item.classList.remove("active");
                    const otherBtn = item.querySelector(".resto-menu-status");
                    if(otherBtn) otherBtn.textContent = "Lihat";
                }
            });

            const isActive = currentItem.classList.toggle("active");
            statusLabel.textContent = isActive ? "Tutup" : "Lihat";
        });
    });
});
